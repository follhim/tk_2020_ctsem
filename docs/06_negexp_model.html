6.2 Two-Part Approach (Censored Magnitude)

6.2.1 Model Specification

neg3_twopart_ctsem <- "
# Measurement Model
## BINARY
bin_neg3_B =~ esm_neg3_bin
bin_neg3_W =~ esm_neg3_bin

## CONT.
con_neg3_B =~ esm_neg3_con
con_neg3_W =~ esm_neg3_con

# Between/Trend
bin_neg3_B  ~  0*bin_neg3_B
bin_neg3_B ~~  0*bin_neg3_B
bin_neg3_B %~% bin_neg3_B

con_neg3_B  ~  0*con_neg3_B
con_neg3_B ~~  0*con_neg3_B
con_neg3_B %~% con_neg3_B

bin_neg3_B %~% con_neg3_B

# Within (dynamic) factor
## BINARY
bin_neg3_W  ~  bin_neg3_W
bin_neg3_W ~~  bin_neg3_W
bin_neg3_W %~% bin_neg3_W

## CONT.
con_neg3_W  ~  con_neg3_W
con_neg3_W ~~  con_neg3_W
con_neg3_W %~% con_neg3_W

## T0MEAN
bin_neg3_B ~t0~ 1
con_neg3_B ~t0~ 1

BINARY: esm_neg3_bin
CENSOR: esm_neg3_con
"

neg3_twopart_syn <- ct_lavaan(neg3_twopart_ctsem, data = p6_data)

neg3_twopart_mod <- ctModel(
  type          = "ct",
  id            = "id",
  time          = "time_day",
  latentNames   = neg3_twopart_syn$latentNames,
  manifestNames = neg3_twopart_syn$manifestNames,
  LAMBDA        = neg3_twopart_syn$LAMBDA,
  DRIFT         = neg3_twopart_syn$DRIFT,
  DIFFUSION     = neg3_twopart_syn$DIFFUSION,
  T0VAR         = neg3_twopart_syn$T0VAR,
  CINT          = neg3_twopart_syn$CINT,
  T0MEANS       = neg3_twopart_syn$T0MEANS,
  MANIFESTMEANS = neg3_twopart_syn$MANIFESTMEANS,
  manifesttype  = neg3_twopart_syn$manifesttype,
  censormin     = neg3_twopart_syn$censormin,
  censormax     = neg3_twopart_syn$censormax,
  n.TIpred      = 3,
  TIpredNames   = c("zpos", "zneg", "zdis")
)

effect_cols <- c("zpos_effect", "zneg_effect", "zdis_effect")

# Only interested in cross-level moderation of the within-person (dynamic)
# processes' autoregression -- not the between-person/trait processes, and
# not any of the bin<->con cross-latent DRIFT paths (never specified in
# this spec, fixed at 0 by default, but ctModel() still defaults their
# TIpred effects to TRUE regardless of that fixed value). Start by
# switching every parameter's TIpred effects off, then re-enable only the
# two within-person autoregressions.
neg3_twopart_mod$pars[, effect_cols] <- FALSE

bin_w_idx <- match("bin_neg3_W", neg3_twopart_syn$latentNames)
con_w_idx <- match("con_neg3_W", neg3_twopart_syn$latentNames)

target_bin_w <- neg3_twopart_mod$pars$matrix == "DRIFT" &
  as.character(neg3_twopart_mod$pars$row) == as.character(bin_w_idx) &
  as.character(neg3_twopart_mod$pars$col) == as.character(bin_w_idx)
target_con_w <- neg3_twopart_mod$pars$matrix == "DRIFT" &
  as.character(neg3_twopart_mod$pars$row) == as.character(con_w_idx) &
  as.character(neg3_twopart_mod$pars$col) == as.character(con_w_idx)

neg3_twopart_mod$pars[target_bin_w | target_con_w, effect_cols] <- TRUE

# Safety check: ctModel() has been observed to silently default
# indvarying = FALSE on a free DRIFT diagonal parameter (found with
# con_psx_W in 03_ple_model.qmd) even when nothing in this script asked
# for that. Confirm both within-person autoregressions are still
# individually varying, and force them back on if ctModel() defaulted
# either off.
stopifnot("indvarying" %in% names(neg3_twopart_mod$pars))
neg3_twopart_mod$pars[target_bin_w | target_con_w, "indvarying"] <- TRUE

print(neg3_twopart_mod$pars[neg3_twopart_mod$pars$matrix == "DRIFT",
                           c("matrix", "row", "col", "indvarying", effect_cols)])
