6.2 Two-Part Approach (Censored Magnitude)

6.2.1 Model Specification

neg3_twopart_ctsem <- "
# Measurement Model
## BINARY
bin_neg3_B =~ esm_neg3_bin
bin_neg3_W =~ esm_neg3_bin

## CONT.
con_neg3_B =~ esm_neg3_con
con_neg3_W =~ esm_neg3_con

# Between/Trend
bin_neg3_B  ~  bin_neg3_B
bin_neg3_B ~~  0*bin_neg3_B
bin_neg3_B %~% bin_neg3_B

con_neg3_B  ~  con_neg3_B
con_neg3_B ~~  0*con_neg3_B
con_neg3_B %~% con_neg3_B

bin_neg3_B %~% con_neg3_B

# Within (dynamic) factor
## BINARY
bin_neg3_W  ~  bin_neg3_W
bin_neg3_W ~~  bin_neg3_W
bin_neg3_W %~% bin_neg3_W

## CONT.
con_neg3_W  ~  con_neg3_W
con_neg3_W ~~  con_neg3_W
con_neg3_W %~% con_neg3_W

## T0MEAN
bin_neg3_B ~t0~ 1
con_neg3_B ~t0~ 1

BINARY: esm_neg3_bin
CENSOR: esm_neg3_con
"

neg3_twopart_syn <- ct_lavaan(neg3_twopart_ctsem, data = p6_data)

neg3_twopart_mod <- ctModel(
  type          = "ct",
  id            = "id",
  time          = "time_day",
  latentNames   = neg3_twopart_syn$latentNames,
  manifestNames = neg3_twopart_syn$manifestNames,
  LAMBDA        = neg3_twopart_syn$LAMBDA,
  DRIFT         = neg3_twopart_syn$DRIFT,
  DIFFUSION     = neg3_twopart_syn$DIFFUSION,
  T0VAR         = neg3_twopart_syn$T0VAR,
  CINT          = neg3_twopart_syn$CINT,
  T0MEANS       = neg3_twopart_syn$T0MEANS,
  MANIFESTMEANS = neg3_twopart_syn$MANIFESTMEANS,
  manifesttype  = neg3_twopart_syn$manifesttype,
  censormin     = neg3_twopart_syn$censormin,
  censormax     = neg3_twopart_syn$censormax,
  n.TIpred      = 3,
