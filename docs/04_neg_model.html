4.2 Two-Part Approach (Censored Magnitude)

4.2.1 Model Specification

na_twopart_ctsem <- "
# Measurement Model
## BINARY
bin_na_B =~ esm_na_bin
bin_na_W =~ esm_na_bin

## CONT.
con_na_B =~ esm_na_con
con_na_W =~ esm_na_con

# Between/Trend
bin_na_B  ~  bin_na_B
bin_na_B ~~  0*bin_na_B
bin_na_B %~% bin_na_B

con_na_B  ~  con_na_B
con_na_B ~~  0*con_na_B
con_na_B %~% con_na_B

bin_na_B %~% con_na_B

# Within (dynamic) factor
## BINARY
bin_na_W  ~  bin_na_W
bin_na_W ~~  bin_na_W
bin_na_W %~% bin_na_W

## CONT.
con_na_W  ~  con_na_W
con_na_W ~~  con_na_W
con_na_W %~% con_na_W

## T0MEAN
bin_na_B ~t0~ 1
con_na_B ~t0~ 1

BINARY: esm_na_bin
CENSOR: esm_na_con
"

na_twopart_syn <- ct_lavaan(na_twopart_ctsem, data = p4_data)

na_twopart_mod <- ctModel(
  type          = "ct",
  id            = "id",
  time          = "time_day",
  latentNames   = na_twopart_syn$latentNames,
  manifestNames = na_twopart_syn$manifestNames,
  LAMBDA        = na_twopart_syn$LAMBDA,
  DRIFT         = na_twopart_syn$DRIFT,
  DIFFUSION     = na_twopart_syn$DIFFUSION,
  T0VAR         = na_twopart_syn$T0VAR,
  CINT          = na_twopart_syn$CINT,
  T0MEANS       = na_twopart_syn$T0MEANS,
  MANIFESTMEANS = na_twopart_syn$MANIFESTMEANS,
  manifesttype  = na_twopart_syn$manifesttype,
  censormin     = na_twopart_syn$censormin,
  censormax     = na_twopart_syn$censormax,
  n.TIpred      = 3,
  TIpredNames   = c("zpos", "zneg", "zdis")
)
