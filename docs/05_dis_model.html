5.2 Two-Part Approach (Censored Magnitude)

5.2.1 Model Specification

dis_twopart_ctsem <- "
# Measurement Model
## BINARY
bin_dis_B =~ esm_dis_bin
bin_dis_W =~ esm_dis_bin

## CONT.
con_dis_B =~ esm_dis_con
con_dis_W =~ esm_dis_con

# Between
bin_dis_B  ~  0*bin_dis_B
bin_dis_B ~~  0*bin_dis_B
bin_dis_B %~% bin_dis_B

con_dis_B  ~  0*con_dis_B
con_dis_B ~~  0*con_dis_B
con_dis_B %~% con_dis_B

bin_dis_B %~% con_dis_B

# Within (dynamic) factor
## BINARY
bin_dis_W  ~  bin_dis_W
bin_dis_W ~~  bin_dis_W
bin_dis_W %~% bin_dis_W

## CONT.
con_dis_W  ~  con_dis_W
con_dis_W ~~  con_dis_W
con_dis_W %~% con_dis_W

## T0MEAN
bin_dis_B ~t0~ 1
con_dis_B ~t0~ 1

BINARY: esm_dis_bin
CENSOR: esm_dis_con
"

dis_twopart_syn <- ct_lavaan(dis_twopart_ctsem, data = p5_data)

dis_twopart_mod <- ctModel(
  type          = "ct",
  id            = "id",
  time          = "time_day",
  latentNames   = dis_twopart_syn$latentNames,
  manifestNames = dis_twopart_syn$manifestNames,
  LAMBDA        = dis_twopart_syn$LAMBDA,
  DRIFT         = dis_twopart_syn$DRIFT,
  DIFFUSION     = dis_twopart_syn$DIFFUSION,
  T0VAR         = dis_twopart_syn$T0VAR,
  CINT          = dis_twopart_syn$CINT,
  T0MEANS       = dis_twopart_syn$T0MEANS,
  MANIFESTMEANS = dis_twopart_syn$MANIFESTMEANS,
  manifesttype  = dis_twopart_syn$manifesttype,
  censormin     = dis_twopart_syn$censormin,
  censormax     = dis_twopart_syn$censormax,
  n.TIpred      = 3,
  TIpredNames   = c("zpos", "zneg", "zdis")
)

effect_cols <- c("zpos_effect", "zneg_effect", "zdis_effect")

# Only interested in cross-level moderation of the within-person (dynamic)
# processes' autoregression -- not the between-person/trait processes, and
# not any of the bin<->con cross-latent DRIFT paths (never specified in
# this spec, fixed at 0 by default, but ctModel() still defaults their
# TIpred effects to TRUE regardless of that fixed value). Start by
# switching every parameter's TIpred effects off, then re-enable only the
# two within-person autoregressions.
dis_twopart_mod$pars[, effect_cols] <- FALSE

bin_w_idx <- match("bin_dis_W", dis_twopart_syn$latentNames)
con_w_idx <- match("con_dis_W", dis_twopart_syn$latentNames)

target_bin_w <- dis_twopart_mod$pars$matrix == "DRIFT" &
  as.character(dis_twopart_mod$pars$row) == as.character(bin_w_idx) &
  as.character(dis_twopart_mod$pars$col) == as.character(bin_w_idx)
target_con_w <- dis_twopart_mod$pars$matrix == "DRIFT" &
  as.character(dis_twopart_mod$pars$row) == as.character(con_w_idx) &
  as.character(dis_twopart_mod$pars$col) == as.character(con_w_idx)

dis_twopart_mod$pars[target_bin_w | target_con_w, effect_cols] <- TRUE


stopifnot("indvarying" %in% names(dis_twopart_mod$pars))
dis_twopart_mod$pars[target_bin_w | target_con_w, "indvarying"] <- TRUE

print(dis_twopart_mod$pars[dis_twopart_mod$pars$matrix == "DRIFT",
